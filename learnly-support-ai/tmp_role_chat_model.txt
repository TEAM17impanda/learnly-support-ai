RoleChatMessage(BaseModel):

    scenario_id: str | None = None
    message: str

    @field_validator("message")
    @classmethod
    def role_message_is_not_blank(cls, value: str) -> str:

        if not value:
            raise ValueError("message must not be blank")
        return value
